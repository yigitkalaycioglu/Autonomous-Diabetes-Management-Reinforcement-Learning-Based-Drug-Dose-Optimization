print("\n--- 5. Korelasyon Analizi ---")

# Kategorik verileri sayısal hale getir (sadece bu grafik için geçici kopyada)
df_corr = df.copy()
le = LabelEncoder()
for col in df_corr.select_dtypes(include='object').columns:
    df_corr[col] = le.fit_transform(df_corr[col])

plt.figure(figsize=(12, 10))
correlation_matrix = df_corr.corr()
sns.heatmap(correlation_matrix, annot=True, cmap='coolwarm', fmt=".2f", linewidths=0.5)
plt.title('Özellikler Arası Korelasyon Matrisi')
plt.show()

--- 5. Korelasyon Analizi ---

'blood_glucose_level' ile 'diabetes' (0.42) ve 'HbA1c_level' (0.17) arasında pozitif korelasyon vardır. Bu biyolojik olarak beklenen bir durumdur.

print("\n--- 6. Klinik Risk ve Güvenlik Analizi (RL Hazırlığı) ---")

plt.figure(figsize=(14, 6))
# Şeker Dağılımı ve Kritik Bölgeler
sns.kdeplot(df['blood_glucose_level'], fill=True, color='blue', alpha=0.3)
plt.axvline(x=70, color='red', linestyle='--', label='Hipoglisemi (<70)')
plt.axvline(x=180, color='orange', linestyle='--', label='Hiperglisemi (>180)')
plt.axvspan(0, 70, color='red', alpha=0.1)
plt.axvspan(100, 140, color='green', alpha=0.1, label='Hedef Aralık')

plt.title("Hasta Popülasyonu ve Klinik Güvenlik Bölgeleri")
plt.xlabel("Kan Şekeri (mg/dL)")
plt.legend()
plt.xlim(0, 400)
plt.show()
